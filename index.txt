  <script>
    function createHearts() {
      const symbols = ['❤️', '💖', '✨', '🌸', '💫'];
      for (let i = 0; i < 22; i++) {
        const heart = document.createElement('div');
        heart.classList.add('heart-bg');
        heart.innerText = symbols[Math.floor(Math.random() * symbols.length)];
        heart.style.left = Math.random() * 100 + 'vw';
        heart.style.animationDuration = (Math.random() * 3 + 3) + 's';
        heart.style.animationDelay = (Math.random() * 3) + 's';
        heart.style.fontSize = (Math.random() * 15 + 14) + 'px';
        document.body.appendChild(heart);
      }
    }
    createHearts();

    function revealSurprise() {
      const song = document.getElementById('bgSong');
      song.play().catch(() => console.log('Autoplay issue'));

      document.getElementById('introScreen').style.display = 'none';
      const surprise = document.getElementById('surpriseScreen');
      surprise.style.display = 'block';
      setTimeout(() => surprise.style.opacity = '1', 50);

      startTypewriter();
      startAutoSlide();
    }

    // Slider Logic
    let currentSlide = 0;
    let slideTimer;

    function showSlide(index) {
      const slides = document.querySelectorAll('.slide');
      if (index >= slides.length) currentSlide = 0;
      else if (index < 0) currentSlide = slides.length - 1;
      else currentSlide = index;

      slides.forEach((s, i) => {
        s.classList.toggle('active', i === currentSlide);
      });
    }

    function changeSlide(step) {
      showSlide(currentSlide + step);
      resetSlideTimer();
    }

    function startAutoSlide() {
      slideTimer = setInterval(() => {
        showSlide(currentSlide + 1);
      }, 3000); // Har 3 second me photo change hogi
    }

    function resetSlideTimer() {
      clearInterval(slideTimer);
      startAutoSlide();
    }

    // Typewriter
    const message = "Aapki smile aur aapki baatein meri duniya roshan kar deti hain. Har pal aapke sath behad khubsurat hai...";
    let index = 0;

    function startTypewriter() {
      if (index < message.length) {
        document.getElementById('typedMessage').innerHTML += message.charAt(index);
        index++;
        setTimeout(startTypewriter, 50);
      }
    }
